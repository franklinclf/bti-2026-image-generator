```ts
// Icone "in memoriam": aureola de circuito (aureola + no + duas trilhas como asas).
// Desenhado em unidades de em, baseline em y = 0, y para baixo. Preenchimento nonzero:
// o anel da aureola usa elipse interna no sentido oposto.
import type { Laid } from './layout';

const r2 = (v: number) => Math.round(v * 100) / 100;

function ellipse(cx: number, cy: number, rx: number, ry: number, sweep: 0 | 1, k: number): string {
  const x0 = r2((cx - rx) * k);
  const y = r2(cy * k);
  const a = `${r2(rx * k)},${r2(ry * k)} 0 1,${sweep}`;
  return `M${x0},${y} a${a} ${r2(2 * rx * k)},0 a${a} ${r2(-2 * rx * k)},0 Z`;
}

function poly(pts: [number, number][], k: number): string {
  return 'M' + pts.map(([x, y]) => `${r2(x * k)},${r2(y * k)}`).join(' L') + ' Z';
}

export const MEMORIAM_ADV = 0.75; // avanco em em

export function memoriamIcon(sizeMm: number): Laid {
  const k = sizeMm;
  const left: [number, number][] = [[0.26, -0.40], [0.07, -0.52], [0.03, -0.64], [0.08, -0.64], [0.11, -0.56], [0.27, -0.46]];
  const right = left.map(([x, y]) => [0.7 - x, y] as [number, number]).reverse();
  const d = [
    ellipse(0.35, -0.80, 0.22, 0.07, 0, k), // aureola (externa)
    ellipse(0.35, -0.80, 0.17, 0.035, 1, k), // furo da aureola (sentido oposto)
    ellipse(0.35, -0.38, 0.12, 0.12, 0, k), // no
    poly(left, k),
    poly(right, k),
    ellipse(0.055, -0.66, 0.035, 0.035, 0, k), // pads nas pontas
    ellipse(0.645, -0.66, 0.035, 0.035, 0, k),
  ].join('');
  return { d, width: MEMORIAM_ADV * sizeMm };
}
```

### Task 4: More flexible portrait matching

**Files:** `src/mural/photos.ts`, `src/mural/photos.test.ts`. Keep the exported signature `matchPortraits(formandos, fileNames): MatchResult` and result ordering (matches by fileIndex asc, unmatched asc).

Algorithm:
- `tokensOf(s)`: `normalizeName(s)` split on spaces, then drop: stopwords `de da do das dos e`; noise words `img dsc dscn pxl foto photo retrato final edit editada editado copia copy formando formanda` ; any token containing a digit.
- Token match between a file token `t` and a name token `n`: **exact** (`t === n`, weight 3); **prefix** (`t.length >= 3 && n.startsWith(t)`, weight 2); **typo** (`t.length >= 5 && n.length >= 5 && levenshtein(t, n) === 1`, weight 1). Each name token can be used once per file (greedy: try the file tokens in order, pick the best-weight unused name token).
- A formando is a candidate for a file when every file token matches; score = sum of weights. Word order doesn't matter.
- Passes, each over the files still unmatched and formandos still free:
  1. exact full normalized name (as today, using `normalizeName` without dropping tokens);
  2. candidates with ≥ 2 file tokens: assign when there is a single best score (ties → leave for later);
  3. single file token (length ≥ 4): assign when exactly one free formando is a candidate.
- Files whose token list is empty stay unmatched. Write `levenshtein` as a small local function.
